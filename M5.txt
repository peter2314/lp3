

import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering
from sklearn.metrics import silhouette_score
from scipy.cluster.hierarchy import dendrogram, linkage

# Load dataset
data = pd.read_csv("sales_data_sample.csv", encoding='latin1')

# Select numerical columns
X = data.select_dtypes(include=['number'])

# Remove missing values
X = X.fillna(X.median())

# Scale the data
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# K-Means Clustering
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
data['KMeans_Cluster'] = kmeans.fit_predict(X_scaled)

print("K-Means Clustering")
print(data['KMeans_Cluster'])

print("K-Means Silhouette Score:",
      silhouette_score(X_scaled, data['KMeans_Cluster']))

# Hierarchical Clustering
hierarchical = AgglomerativeClustering(n_clusters=3)
data['Hierarchical_Cluster'] = hierarchical.fit_predict(X_scaled)

print("\nHierarchical Clustering")
print(data['Hierarchical_Cluster'])

print("Hierarchical Silhouette Score:",
      silhouette_score(X_scaled, data['Hierarchical_Cluster']))

# Display Dendrogram using a sample
sample = X_scaled[:100]
linked = linkage(sample, method='ward')

plt.figure(figsize=(10, 5))
dendrogram(linked)
plt.title("Hierarchical Clustering Dendrogram")
plt.xlabel("Data Points")
plt.ylabel("Distance")
plt.show()
